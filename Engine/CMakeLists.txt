# Layers below the engine are separate libraries, so the import helper links no GPU code at all.
file(GLOB base_sources CONFIGURE_DEPENDS base/*.cpp)
add_library(geo_base STATIC ${base_sources})
target_include_directories(geo_base PUBLIC .)
target_link_libraries(geo_base PUBLIC geo_options PRIVATE geo_warnings)

add_library(geo_format STATIC "${CMAKE_SOURCE_DIR}/Import/format/blob.cpp")
target_include_directories(geo_format PUBLIC "${CMAKE_SOURCE_DIR}/Import")
target_link_libraries(geo_format PUBLIC geo_base PRIVATE geo_warnings)

# The engine's logic and the layers under it, the same on every platform: no Metal, no threads of
# its own, no processes. The Mac engine and the universal viewer each add a platform around it.
file(GLOB_RECURSE engine_core_sources CONFIGURE_DEPENDS
    math/*.cpp nav/*.cpp scene/*.cpp model/*.cpp render/*.cpp core/*.cpp)
add_library(engine_core STATIC ${engine_core_sources})
target_include_directories(engine_core PUBLIC include PUBLIC .)
target_include_directories(engine_core SYSTEM PRIVATE "${CMAKE_SOURCE_DIR}/ThirdParty/xxhash")
target_link_libraries(engine_core PUBLIC geo_base geo_format PRIVATE geo_warnings)

# The WebGPU renderer, with its WGSL embedded as strings. Editing a shader reconfigures.
if(GEO_WEBGPU)
    set(wgsl_cpp "${CMAKE_CURRENT_BINARY_DIR}/wgsl_sources.cpp")
    set(wgsl_text "// Generated by Engine/CMakeLists.txt from Engine/webgpu/shaders. Edit those.\n")
    string(APPEND wgsl_text "#include \"webgpu/wgsl.h\"\n\nnamespace geo::webgpu::wgsl {\n\n")
    foreach(name common cull surface edges sheet post overlay)
        set(wgsl_file "${CMAKE_CURRENT_SOURCE_DIR}/webgpu/shaders/${name}.wgsl")
        set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${wgsl_file}")
        file(READ "${wgsl_file}" wgsl_source)
        string(SUBSTRING "${name}" 0 1 head)
        string(SUBSTRING "${name}" 1 -1 tail)
        string(TOUPPER "${head}" head)
        string(APPEND wgsl_text "const char* const k${head}${tail} = R\"wgsl(${wgsl_source})wgsl\";\n\n")
    endforeach()
    string(APPEND wgsl_text "} // namespace geo::webgpu::wgsl\n")
    file(CONFIGURE OUTPUT "${wgsl_cpp}" CONTENT "${wgsl_text}" @ONLY)

    add_library(engine_webgpu STATIC webgpu/renderer.cpp "${wgsl_cpp}")
    target_include_directories(engine_webgpu PUBLIC .)
    target_link_libraries(engine_webgpu PUBLIC engine_core PRIVATE geo_warnings)
    if(GEO_TARGET STREQUAL "wasm")
        target_compile_options(engine_webgpu PUBLIC --use-port=emdawnwebgpu)
        target_link_options(engine_webgpu PUBLIC --use-port=emdawnwebgpu)
    else()
        find_package(Dawn CONFIG REQUIRED PATHS "${CMAKE_SOURCE_DIR}/ThirdParty/.dawn/install"
                     NO_DEFAULT_PATH)
        target_link_libraries(engine_webgpu PUBLIC dawn::webgpu_dawn)
    endif()
endif()

if(GEO_TARGET STREQUAL "wasm")
    # The engine in the browser: the core, the WebGPU renderer and the page's host, as a module
    # the viewer's page loads. See browser/web_engine.cpp.
    if(GEO_WEBGPU)
        add_executable(geometryview-web browser/web_engine.cpp)
        target_link_libraries(geometryview-web PRIVATE engine_webgpu engine_core geo_warnings)
        set_target_properties(geometryview-web PROPERTIES SUFFIX ".mjs")
        target_link_options(geometryview-web PRIVATE
            -sALLOW_MEMORY_GROWTH=1 -sMAXIMUM_MEMORY=4GB
            -sEXPORTED_FUNCTIONS=_malloc,_free
            -sEXPORTED_RUNTIME_METHODS=HEAPU8,HEAPU32,UTF8ToString,stringToUTF8,lengthBytesUTF8
            -sENVIRONMENT=web -sMODULARIZE=1 -sEXPORT_ES6=1 -sEXPORT_NAME=createGeometryView)
    endif()
    return()
endif()

    PUBLIC engine_core geo_base geo_format

    "-framework Metal"

    "-framework ImageIO"
